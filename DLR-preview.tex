\begin{frame}[t]{Equations and scientific content}
\textbf{Vector autoregression}\par\medskip
A multivariate model predicts the current observations from their past:
\[
 \mathbf z_t = \sum_{\ell=1}^{p}\mathbf M_{\ell}\mathbf z_{t-\ell}+\mathbf u_t .
\]
\begin{itemize}\setlength\itemsep{2mm}
\item $\mathbf z_t$: the vector of observations at time $t$.
\item $\mathbf M_{\ell}$: coefficients at lag $\ell$.
\item $\mathbf u_t$: the remaining innovations.
\end{itemize}
\end{frame}
